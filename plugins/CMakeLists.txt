list(APPEND CMAKE_MODULE_PATH ${CMAKE_CURRENT_SOURCE_DIR}/cmake)

# ENABLE_PLUGIN (below) calls target_link_libraries(plugin_${plugin} ...) from
# this file's directory scope, but plugin_${plugin} targets are created inside
# add_subdirectory(${plugin}) -- a different directory scope. CMake disallows
# that without CMP0079 NEW. cmake/wayland_cxx.cmake sets this too, but policy
# settings don't propagate across sibling add_subdirectory() scopes, only down
# into children of wherever they're set -- set it here too.
cmake_policy(SET CMP0079 NEW)

include_directories(
        ${CMAKE_CURRENT_SOURCE_DIR}/..
        ${PROJECT_BINARY_DIR}/config
        # ivi-homescreen shell root, so plugins resolve the ihs logging shim
        # "logging/logging.h" (ihs::log:: + IHS_DEBUG / IHS_TRACE).
        ${CMAKE_SOURCE_DIR}/shell
        # wayland-cxx-scanner-generated protocol headers (wayland_client.hpp,
        # input_timestamps_client.hpp, etc.) that shell/wayland/*.h pull in --
        # generated by the shell's own ivi_wayland_protocols(${PROJECT_NAME})
        # call (shell/CMakeLists.txt) into this one flat directory regardless
        # of which specific header. Build-order dependency against the
        # wlcxxgen_* custom targets is wired generically in the top-level
        # CMakeLists.txt (after add_subdirectory(shell)), not here.
        ${PROJECT_BINARY_DIR}/shell/wayland-protocols
)

# This plugins tree's common/ code calls ihs::log::{info,warn,error,...} and
# IHS_DEBUG/IHS_TRACE, provided by shell/logging/logging.h -- but only on
# shell pins past its spdlog-to-ihs_shared rewrite. Older pins (this one
# included) still expose spdlog directly (spdlog::info/warn/error/... +
# LOG_*/DLOG_* macros, see shell/logging/logging.h). Detect via a plain text
# search (a real compile probe would need the same include/definition set as
# the actual target, which is fragile to replicate here) and, if absent,
# define IHS_LOG_SHIM so plugins/common/logging.h maps ihs::log:: straight
# onto spdlog:: (identical fmt-style call signatures) instead of hard-failing.
file(READ ${CMAKE_SOURCE_DIR}/shell/logging/logging.h _ihs_logging_h)
string(FIND "${_ihs_logging_h}" "namespace ihs::log" _ihs_log_ns_pos)
if (_ihs_log_ns_pos EQUAL -1)
    add_compile_definitions(IHS_LOG_SHIM=1)
    # On this older codepath, shell/logging/logging.h #includes real spdlog
    # headers unconditionally (spdlog/sinks/stdout_color_sinks.h etc.), so
    # anything transitively including it needs spdlog's own include root too.
    # third_party/CMakeLists.txt only exposes this via the `spdlog` INTERFACE
    # target's own compile options (-isystem.../third_party/spdlog/include),
    # and no plugin here links that target -- add the path directly instead.
    include_directories(${CMAKE_SOURCE_DIR}/third_party/spdlog/include)
endif ()

macro(ENABLE_PLUGIN plugin)

    string(TOUPPER ${plugin} ucase_plugin)

    set(ENABLE_PLUGIN_${ucase_plugin} ON)

    if (${plugin} MATCHES ".*_view")
        message(STATUS "View: ${plugin}")
    else ()
        message(STATUS "Plugin: ${plugin}")
    endif ()

    add_subdirectory(${plugin})

    # shell/wayland/*.h (pulled in transitively by any plugin touching the
    # shell's Wayland layer) needs the wl/ header-only framework the shell
    # itself gets via ivi_wayland_link() (cmake/wayland_cxx.cmake) -- plugins
    # never called that. chromium_dart_view already links this explicitly
    # itself; wire it for every plugin here instead of one at a time.
    if (TARGET wayland-cxx::wayland-cxx)
        target_link_libraries(plugin_${plugin} PRIVATE wayland-cxx::wayland-cxx)
    endif ()

    target_compile_definitions(plugin_${plugin} PUBLIC ENABLE_PLUGIN_${ucase_plugin})

    add_sanitizers(plugin_${plugin})

    if (IPO_SUPPORT_RESULT)
        set_property(TARGET plugin_${plugin} PROPERTY INTERPROCEDURAL_OPTIMIZATION TRUE)
    endif ()


    target_link_libraries(plugins PUBLIC plugin_${plugin})

    target_compile_definitions(plugins PUBLIC ENABLE_PLUGIN_${ucase_plugin})

    #print_target_properties(plugin_${plugin})

endmacro(ENABLE_PLUGIN)

macro(PLUGIN_OPTION plugin description default)

    string(TOUPPER ${plugin} ucase_plugin)

    if (ENABLE_PLUGINS)
        option(BUILD_PLUGIN_${ucase_plugin} ${description} ${default})
    else ()
        set(BUILD_PLUGIN_${ucase_plugin} OFF)
    endif ()

    set(ENABLE_PLUGIN_${ucase_plugin} OFF)
    if (BUILD_PLUGIN_${ucase_plugin})
        ENABLE_PLUGIN(${plugin})
    endif ()

endmacro(PLUGIN_OPTION)
